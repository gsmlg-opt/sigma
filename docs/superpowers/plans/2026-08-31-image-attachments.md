> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade DuskMoon Elements and deliver image-only chat attachments from the browser through Sigma's journal and LLM providers back into the persisted user bubble.

**Architecture:** Keep the DuskMoon chat input and extend its existing LiveView hook. A pure browser module validates and serializes `File` objects; a pure Elixir module validates decoded bytes and creates canonical `%{type: :image}` blocks. Existing Agent and JSONL seams carry those blocks unchanged, while Anthropic and OpenAI adapters translate them only at their HTTP boundaries.

**Tech Stack:** Elixir/OTP 28, Phoenix LiveView, `phoenix_duskmoon`, DuskMoon Elements 1.7.4, JavaScript/Bun tests, ExUnit, JSONL session storage, Chrome DevTools MCP.

- [ ] **Step 3: Implement the pure serializer**

```javascript
const SUPPORTED_TYPES = new Set([
  "image/png",
  "image/jpeg",
  "image/gif",
  "image/webp"
])

const MAX_FILES = 4
const MAX_FILE_BYTES = 5 * 1024 * 1024
const MAX_TOTAL_BYTES = 10 * 1024 * 1024

export function validateImageFiles(files) {
  const list = Array.from(files || [])

  if (list.length > MAX_FILES) return "too_many"
  if (list.some((file) => !SUPPORTED_TYPES.has(file.type))) return "unsupported_type"
  if (list.some((file) => file.size > MAX_FILE_BYTES)) return "too_large"

  const total = list.reduce((sum, file) => sum + file.size, 0)
  return total > MAX_TOTAL_BYTES ? "too_large" : null
}

function readAsDataURL(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(reader.result)
    reader.onerror = () => reject(reader.error || new Error("read failed"))
    reader.readAsDataURL(file)
  })
}

function imageFromDataURL(file, url) {
  const prefix = `data:${file.type};base64,`
  if (typeof url !== "string" || !url.startsWith(prefix)) throw new Error("invalid data URL")
  return { data: url.slice(prefix.length), mime_type: file.type }
}

export async function encodeImageFiles(files, read = readAsDataURL) {
  const list = Array.from(files || [])
  const code = validateImageFiles(list)
  if (code) return { ok: false, code }

  try {
    const images = await Promise.all(
      list.map(async (file) => imageFromDataURL(file, await read(file)))
    )
    return { ok: true, images }
  } catch (_error) {
    return { ok: false, code: "read_failed" }
  }
}
```

Implement this public contract and keep all parsing private:

```elixir
defmodule Sigma.Web.ImageAttachments do
  @moduledoc false

  @allowed_mime_types ~w(image/png image/jpeg image/gif image/webp)
  @max_images 4
  @max_image_bytes 5 * 1024 * 1024
  @max_total_bytes 10 * 1024 * 1024

  @client_messages %{
    "unsupported_type" => "Attach PNG, JPEG, single-frame GIF, or WebP images only.",
    "too_many" => "Attach no more than four images.",
    "too_large" => "Images must be at most 5 MiB each and 10 MiB total.",
    "read_failed" => "Sigma could not read one of the attached images.",
    "slash_command" => "Attachments cannot be combined with slash commands."
  }

  @server_messages %{
    unsupported_type: "Attach PNG, JPEG, single-frame GIF, or WebP images only.",
    too_many: "Attach no more than four images.",
    too_large: "Images must be at most 5 MiB each and 10 MiB total.",
    invalid_data: "One of the attached images is invalid.",
    animated_gif: "Animated GIF attachments are not supported.",
    slash_command: "Attachments cannot be combined with slash commands."
  }

  def normalize(text, raw_images) when is_binary(text) and is_list(raw_images) do
    text = String.trim(text)

    with :ok <- validate_count(raw_images),
         :ok <- validate_slash_command(text, raw_images),
         {:ok, images} <- decode_images(raw_images) do
      build_content(text, images)
    end
  end

  def normalize(_text, _images), do: {:error, :invalid_data}

  def client_error(code), do: Map.fetch(@client_messages, code)
  def error_message(reason), do: Map.fetch!(@server_messages, reason)

  def data_url(%{type: :image, data: data, mime_type: mime})
      when mime in @allowed_mime_types and is_binary(data),
      do: "data:#{mime};base64,#{data}"

  def data_url(_), do: nil

  defp validate_count(images) when length(images) <= @max_images, do: :ok
  defp validate_count(_images), do: {:error, :too_many}

  defp validate_slash_command("/" <> _rest, [_ | _]), do: {:error, :slash_command}
  defp validate_slash_command(_text, _images), do: :ok

  defp decode_images(images) do
    Enum.reduce_while(images, {:ok, [], 0}, fn raw, {:ok, acc, total} ->
      case decode_image(raw) do
        {:ok, block, size} when total + size <= @max_total_bytes ->
          {:cont, {:ok, [block | acc], total + size}}

        {:ok, _block, _size} ->
          {:halt, {:error, :too_large}}

        {:error, reason} ->
          {:halt, {:error, reason}}
      end
    end)
    |> case do
      {:ok, blocks, _total} -> {:ok, Enum.reverse(blocks)}
      error -> error
    end
  end

  defp decode_image(%{"mime_type" => mime, "data" => data})
       when mime in @allowed_mime_types and is_binary(data) do
    with {:ok, bytes} <- Base.decode64(data),
         :ok <- validate_size(bytes),
         :ok <- validate_signature(mime, bytes),
         :ok <- validate_animation(mime, bytes) do
      {:ok, %{type: :image, data: data, mime_type: mime}, byte_size(bytes)}
    else
      :error -> {:error, :invalid_data}
      {:error, _reason} = error -> error
    end
  end

  defp decode_image(%{"mime_type" => _mime, "data" => _data}),
    do: {:error, :unsupported_type}

  defp decode_image(_raw), do: {:error, :invalid_data}

  defp validate_size(bytes) when byte_size(bytes) <= @max_image_bytes, do: :ok
  defp validate_size(_bytes), do: {:error, :too_large}

  defp validate_signature("image/png", <<137, 80, 78, 71, 13, 10, 26, 10, _::binary>>), do: :ok
  defp validate_signature("image/jpeg", <<255, 216, 255, _::binary>>), do: :ok
  defp validate_signature("image/gif", <<header::binary-size(6), _::binary>>)
       when header in ["GIF87a", "GIF89a"], do: :ok
  defp validate_signature("image/webp", <<"RIFF", _::little-32, "WEBP", _::binary>>), do: :ok
  defp validate_signature(_mime, _bytes), do: {:error, :invalid_data}

  defp validate_animation("image/gif", bytes) do
    case gif_frame_count(bytes) do
      {:ok, 1} -> :ok
      {:ok, count} when count > 1 -> {:error, :animated_gif}
      _ -> {:error, :invalid_data}
    end
  end

  defp validate_animation(_mime, _bytes), do: :ok

  defp gif_frame_count(
         <<header::binary-size(6), _width::little-16, _height::little-16,
           packed, _background, _aspect, rest::binary>>
       ) when header in ["GIF87a", "GIF89a"] do
    with {:ok, rest} <- skip_color_table(rest, packed),
         do: count_gif_blocks(rest, 0)
  end

  defp gif_frame_count(_bytes), do: {:error, :invalid_data}

  defp skip_color_table(rest, packed) do
    if Bitwise.band(packed, 0x80) == 0 do
      {:ok, rest}
    else
      bytes = 3 * Bitwise.bsl(1, Bitwise.band(packed, 0x07) + 1)
      case rest do
        <<_table::binary-size(bytes), tail::binary>> -> {:ok, tail}
        _ -> {:error, :invalid_data}
      end
    end
  end

  defp count_gif_blocks(<<0x3B, _::binary>>, frames), do: {:ok, frames}

  defp count_gif_blocks(<<0x21, _label, rest::binary>>, frames) do
    with {:ok, rest} <- skip_sub_blocks(rest),
         do: count_gif_blocks(rest, frames)
  end

  defp count_gif_blocks(
         <<0x2C, _left::little-16, _top::little-16, _width::little-16,
           _height::little-16, packed, rest::binary>>,
         frames
       ) do
    with {:ok, rest} <- skip_color_table(rest, packed),
         <<_lzw_code_size, data::binary>> <- rest,
         {:ok, rest} <- skip_sub_blocks(data) do
      next_frames = frames + 1
      if next_frames > 1, do: {:ok, next_frames}, else: count_gif_blocks(rest, next_frames)
    else
      _ -> {:error, :invalid_data}
    end
  end

  defp count_gif_blocks(_rest, _frames), do: {:error, :invalid_data}

  defp skip_sub_blocks(<<0, rest::binary>>), do: {:ok, rest}
  defp skip_sub_blocks(<<size, _data::binary-size(size), rest::binary>>),
    do: skip_sub_blocks(rest)
  defp skip_sub_blocks(_rest), do: {:error, :invalid_data}

  defp build_content("", []), do: {:ok, :empty}
  defp build_content(text, []), do: {:ok, text}
  defp build_content("", images), do: {:ok, images}
  defp build_content(text, images),
    do: {:ok, [%{type: :text, text: text} | images]}
end
```
